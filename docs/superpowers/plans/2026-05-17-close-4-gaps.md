```python
async def _do_refresh(self) -> None:
    if AsyncSessionLocal is None:
        return

    loop = asyncio.get_event_loop()
    now = datetime.now(timezone.utc)

    # Get watched tables from DB (fall back to GIGLER_TABLES for first run)
    async with AsyncSessionLocal() as db:
        from sqlalchemy import select as sa_select
        from dqt_server.models.gigler import Dataset as DatasetModel, ColumnCheck
        result = await db.execute(
            sa_select(DatasetModel).where(DatasetModel.source_id == GIGLER_SOURCE_ID)
        )
        watched = [d.id for d in result.scalars().all()]
        if not watched:
            watched = list(GIGLER_TABLES)

        # Load all user-defined checks for this source's tables
        checks_result = await db.execute(
            sa_select(ColumnCheck).where(ColumnCheck.dataset_id.in_(watched))
        )
        user_checks: list[ColumnCheck] = list(checks_result.scalars().all())

    # Group user checks by table
    checks_by_table: dict[str, list] = {t: [] for t in watched}
    for c in user_checks:
        if c.dataset_id in checks_by_table:
            checks_by_table[c.dataset_id].append({
                "column": c.column_name,
                "slug": c.detector_slug,
                "params": c.params or {},
            })

    # Upsert source record
    async with AsyncSessionLocal() as db:
        stmt = pg_insert(Source).values(
            id=GIGLER_SOURCE_ID,
            name="Gigler ClickHouse",
            engine="ClickHouse",
            host=os.environ.get("CLICKHOUSE_URL", "unknown"),
            port=int(os.environ.get("CLICKHOUSE_PORT", "443")),
            db_name=await loop.run_in_executor(None, self._find_schema),
            username=os.environ.get("CLICKHOUSE_USER", ""),
            status="unknown",
            table_count=len(watched),
            last_synced_at=now,
            created_at=now,
        ).on_conflict_do_update(
            index_elements=["id"],
            set_={
                "last_synced_at": now,
                "table_count": len(watched),
                "host": os.environ.get("CLICKHOUSE_URL", "unknown"),
            },
        )
        await db.execute(stmt)
        await db.commit()

    all_ok = True
    for table in watched:
        result: TableCheckResult = await loop.run_in_executor(
            None, self._check_table_sync, table
        )
        if result.status == "fail":
            all_ok = False

        # Run user-defined checks for this table
        extra_checks = checks_by_table.get(table, [])
        if extra_checks:
            extra_results = await loop.run_in_executor(
                None, self._run_user_checks_sync, table, result.col_names, extra_checks
            )
            await self._persist_user_check_results(extra_results, table, now)

        await self._persist_table_result(result, now)

    async with AsyncSessionLocal() as db:
        src = await db.get(Source, GIGLER_SOURCE_ID)
        if src:
            src.status = "pass" if all_ok else "warn"
            src.last_synced_at = now
            await db.commit()
```

Add `_run_user_checks_sync`:

```python
def _run_user_checks_sync(
    self, table: str, col_names: list[str],
    checks: list[dict]
) -> list[dict]:
    """Run user-defined detector checks on ClickHouse data."""
    from dqt.algorithms._registry import registry as det_registry
    import pandas as pd

    adapter = self._get_adapter()
    schema = self._find_schema()
    results = []

    for chk in checks:
        col = chk["column"]
        slug = chk["slug"]
        if col not in col_names:
            continue
        if slug == "null_fraction":
            continue  # already handled by _check_table_sync

        try:
            # Fetch column sample (up to 10k rows)
            sql = f"SELECT `{col}` FROM `{schema}`.`{table}` WHERE isNotNull(`{col}`) LIMIT 10000"
            r = adapter._client.query(sql)
            series = pd.Series([row[0] for row in r.result_rows])

            det_class = det_registry.get(slug)
            if det_class is None:
                continue

            det = det_class(**(chk.get("params") or {}))
            state = det.fit(series)
            det_result = det.score(series, state)

            results.append({
                "column": col,
```

```python
async def _real_feed_items(lookback_hours: int, db: AsyncSession) -> list[FeedItem]:
    """Build feed from real check_run history."""
    from dqt_server.api.v1.insights import _get_registry

    registry = _get_registry()
    items: list[FeedItem] = []
    now = datetime.now(timezone.utc)
    window_start = now - timedelta(hours=lookback_hours)
    prior_start = window_start - timedelta(hours=lookback_hours)

    for metric in registry.list():
        table = metric.dataset

        # Latest check_runs in current window
        cur_q = await db.execute(
            select(CheckRun)
            .where(CheckRun.dataset_id == table, CheckRun.ran_at >= window_start)
            .order_by(desc(CheckRun.ran_at))
        )
        cur_runs = cur_q.scalars().all()

        # Prior window
        prior_q = await db.execute(
            select(CheckRun)
            .where(
                CheckRun.dataset_id == table,
                CheckRun.ran_at >= prior_start,
                CheckRun.ran_at < window_start,
            )
            .order_by(desc(CheckRun.ran_at))
        )
        prior_runs = prior_q.scalars().all()

        if not cur_runs:
            continue

        def quality_score(runs: list) -> float:
            if not runs:
                return 0.0
            passing = sum(1 for r in runs if r.verdict == "pass")
            return passing / len(runs)

        cur_score = quality_score(cur_runs)
        prior_score = quality_score(prior_runs) if prior_runs else cur_score
        change = cur_score - prior_score

        # Skip if change is negligible
        if abs(change) < 0.005:
            continue

        significance = min(1.0, abs(change) * 5)
        summary = (
            f"{metric.display_name} {'fell' if change < 0 else 'rose'} "
            f"{abs(change) * 100:.1f}% in the last {lookback_hours}h."
        )

        items.append(FeedItem(
            metric_fqn=metric.fqn,
            display_name=metric.display_name,
            observed_change=change,
            significance=significance,
            executive_tier="finance" in (metric.tags or []),
            novelty=0.7,
            engagement=1.0,
            summary_paragraph=summary,
            primary_channel="data",
            estimated_data_contribution=(abs(change) * 0.5, abs(change)),
            estimated_business_contribution=(0.0, 0.1),
            evidence_chips=[],
        ))

    return items
```
