import time
from typing import Optional
import typer

def run_command(
    manifest: str = typer.Argument(...),
    watch: bool = typer.Option(False, "--watch", help="Re-run on schedule"),
    interval: float = typer.Option(60.0, "--interval", help="Seconds between runs"),
    max_runs: Optional[int] = typer.Option(None, "--max-runs", hidden=True),
) -> None:
    """Run checks from a manifest. --watch re-runs every --interval seconds."""
    runs = 0
    while True:
        _run_once(manifest)
        runs += 1
        if max_runs is not None and runs >= max_runs:
            break
        if not watch:
            break
        typer.echo(f"  next run in {interval:.0f}s — Ctrl+C to stop")
        time.sleep(interval)
```

from __future__ import annotations
import shutil
from pathlib import Path
import typer
import numpy as np
import pandas as pd

_DEMO_DIR = Path("demo")


def seed_command() -> None:
    """Generate synthetic demo data and a starter checks.yaml."""
    rng = np.random.default_rng(42)
    n = 500
    _DEMO_DIR.mkdir(exist_ok=True)

    # fct_orders: lognormal revenue, categorical status, 2% null customer_id
    orders = pd.DataFrame({
        "order_id": range(1, n + 1),
        "amount_usd": rng.lognormal(5.0, 0.5, n).round(2),
        "status": rng.choice(["completed", "cancelled", "pending", "refunded"], n),
        "customer_id": rng.integers(1, 200, n).astype(float),
        "created_at": pd.date_range("2024-01-01", periods=n, freq="1h").astype(str),
    })
    null_idx = rng.choice(n, size=int(n * 0.02), replace=False)
    orders.loc[null_idx, "customer_id"] = None

    # fct_sessions: lognormal duration with heavy tail
    sessions = pd.DataFrame({
        "session_id": range(1, n + 1),
        "duration_s": rng.lognormal(4.0, 1.2, n).round(1),
        "page_views": rng.integers(1, 40, n),
        "user_id": rng.integers(1, 300, n),
    })

    orders.to_csv(_DEMO_DIR / "fct_orders.csv", index=False)
    sessions.to_csv(_DEMO_DIR / "fct_sessions.csv", index=False)

    (_DEMO_DIR / "checks.yaml").write_text("""\
version: "1"
source:
  type: csv
  id: orders
  path: demo/fct_orders.csv
  table_name: fct_orders
checks:
  - schema_name: public
    table_name: fct_orders
    column_name: customer_id
    detector_slug: null_fraction
  - schema_name: public
    table_name: fct_orders

  - schema_name: public

def _get_all_incidents(store: "ResultsStore") -> list[dict]:
    incidents_map: dict = getattr(store, "_incidents", {})
    result = []
    for check_id, inc_list in incidents_map.items():
        for inc in inc_list:
            sev = getattr(inc, "severity", "fail")
            result.append({
                "check_id": str(check_id),
                "incident_id": str(getattr(inc, "incident_id", "")),
                "severity": sev.value if hasattr(sev, "value") else str(sev),
                "score": float(getattr(inc, "score", 0.0)),
                "opened_at": str(getattr(inc, "opened_at", "")),
                "status": getattr(inc, "status", "open"),
                "detector_slug": getattr(inc, "detector_slug", ""),
            })
    return sorted(result, key=lambda r: r["opened_at"], reverse=True)


def _get_incident_detail(store: "ResultsStore", check_id: str) -> dict:
    try:
        from uuid import UUID
        uid = UUID(check_id)
    except ValueError:
        return {"runs": [], "incidents": []}
    runs = []
    incidents = []
    try:

    except Exception:
        pass
    try:
        raw_incidents = store.list_incidents(uid)
        for inc in raw_incidents:
            sev = getattr(inc, "severity", "fail")
            incidents.append({
                "incident_id": str(getattr(inc, "incident_id", "")),
                "severity": sev.value if hasattr(sev, "value") else str(sev),

@app.middleware("http")
async def _token_auth(request: Request, call_next):

from hypothesis import given, settings, HealthCheck
from hypothesis import strategies as st

from dqt.algorithms._registry import registry
from dqt.algorithms._base import Verdict

_MULTIVARIATE = {"isolation_forest_fraction", "mahalanobis_distance", "lof",
                 "hbos", "ecod", "one_class_svm"}
_SKIP = {"bocpd", "matrix_profile", "prophet_anomaly", "callable_check",
         "remote_check", "adwin"}

_SLUGS = [s for s in registry.list_slugs() if s not in _MULTIVARIATE and s not in _SKIP]


@pytest.mark.parametrize("slug", _SLUGS)
@settings(max_examples=15, suppress_health_check=[HealthCheck.too_slow])
@given(
    values=st.lists(
        st.floats(min_value=0.0, max_value=1e6, allow_nan=False, allow_infinity=False),
        min_size=30, max_size=150,
    )
)
def test_invariants(slug, values):
    cls = registry.get(slug)
    detector = cls()
    df = pd.DataFrame({"v": values})
    try:
        state = detector.fit(df)
        result = detector.score(df, state)
    except Exception:
        # Legitimate failures on degenerate input (all-same, etc.) are allowed.
        return
    assert result.verdict in (Verdict.pass_, Verdict.warn, Verdict.fail), \
        f"{slug}: invalid verdict {result.verdict!r}"
    assert isinstance(result.score, float) and np.isfinite(result.score) and result.score >= 0, \
        f"{slug}: invalid score {result.score!r}"
    assert isinstance(result.plain_english, str) and len(result.plain_english.strip()) > 0, \
        f"{slug}: empty plain_english"
    assert isinstance(result.details, dict), \
        f"{slug}: details must be dict, got {type(result.details)}"
```

# benchmarks/fixtures.py
"""Synthetic benchmark fixtures — clean vs anomalous pairs per data shape."""
import numpy as np
import pandas as pd
from dataclasses import dataclass


@dataclass
class BenchmarkFixture:
    name: str
    reference: pd.DataFrame
    current_clean: pd.DataFrame    # should PASS
    current_anomalous: pd.DataFrame  # should WARN or FAIL


def make_fixtures(rng: np.random.Generator) -> list[BenchmarkFixture]:
    n = 500
    return [
        BenchmarkFixture(
            name="normal_mean_shift",
            reference=pd.DataFrame({"v": rng.normal(50, 10, n)}),
            current_clean=pd.DataFrame({"v": rng.normal(50, 10, n)}),
            current_anomalous=pd.DataFrame({"v": rng.normal(80, 10, n)}),
        ),
        BenchmarkFixture(
            name="lognormal_tail_shift",
            reference=pd.DataFrame({"v": rng.lognormal(5.0, 0.5, n)}),
            current_clean=pd.DataFrame({"v": rng.lognormal(5.0, 0.5, n)}),
            current_anomalous=pd.DataFrame({"v": rng.lognormal(5.5, 0.5, n)}),
        ),
        BenchmarkFixture(
            name="outliers_injected_5pct",
            reference=pd.DataFrame({"v": rng.normal(50, 10, n)}),
            current_clean=pd.DataFrame({"v": rng.normal(50, 10, n)}),
            current_anomalous=pd.DataFrame({
                "v": np.concatenate([
                    rng.normal(50, 10, int(n * 0.95)),
                    rng.normal(200, 5, int(n * 0.05)),

        BenchmarkFixture(
            name="nulls_injected_10pct",
            reference=pd.DataFrame({"v": rng.normal(50, 10, n).tolist()}),
            current_clean=pd.DataFrame({"v": rng.normal(50, 10, n).tolist()}),
            current_anomalous=pd.DataFrame({

```python
# benchmarks/run_benchmarks.py
"""
Run all univariate detectors against synthetic fixtures.
Outputs precision/recall/F1 to docs/benchmarks.md.

Usage:
  uv run python benchmarks/run_benchmarks.py
"""
from __future__ import annotations
import sys
from pathlib import Path
import numpy as np

sys.path.insert(0, str(Path(__file__).parent.parent / "packages" / "dqt" / "src"))

from benchmarks.fixtures import make_fixtures
from dqt.algorithms._registry import registry
from dqt.algorithms._base import Verdict

_SKIP = {
    "isolation_forest_fraction", "mahalanobis_distance", "lof", "hbos", "ecod",
    "one_class_svm", "callable_check", "remote_check", "bocpd", "matrix_profile",
    "prophet_anomaly", "adwin",
}


def run_all() -> None:
    rng = np.random.default_rng(0)
    fixtures = make_fixtures(rng)
    rows: list[tuple] = []

    for slug in sorted(registry.list_slugs()):
        if slug in _SKIP:
            continue
        cls = registry.get(slug)
        tp = fp = fn = tn = 0
        for fix in fixtures:
            try:
                det = cls()
                state = det.fit(fix.reference)
                r_clean = det.score(fix.current_clean, state)
                r_anom = det.score(fix.current_anomalous, state)
                if r_clean.verdict == Verdict.pass_:
                    tn += 1
                else:
                    fp += 1
                if r_anom.verdict != Verdict.pass_:
                    tp += 1
                else:
                    fn += 1
            except Exception:
                continue

        total = tp + fp + fn + tn
        if total == 0:
            continue
        precision = tp / (tp + fp) if (tp + fp) > 0 else float("nan")
        recall = tp / (tp + fn) if (tp + fn) > 0 else float("nan")
        f1 = (2 * precision * recall / (precision + recall)
              if (precision + recall) > 0 else float("nan"))
        fpr = fp / (fp + tn) if (fp + tn) > 0 else float("nan")

        def _fmt(v: float) -> str:
            return f"{v:.2f}" if not (v != v) else "—"

        rows.append((slug, tp, fp, fn, tn, _fmt(precision), _fmt(recall), _fmt(f1), _fmt(fpr)))

    header = "| Detector | TP | FP | FN | TN | Precision | Recall | F1 | FPR |\n"
    sep = "|---|---|---|---|---|---|---|---|---|\n"
    body = "".join(
        f"| `{r[0]}` | {r[1]} | {r[2]} | {r[3]} | {r[4]} | {r[5]} | {r[6]} | {r[7]} | {r[8]} |\n"
        for r in rows
    )
    out = (
        "# dqt Detector Benchmarks\n\n"
        "_Auto-generated by `benchmarks/run_benchmarks.py`. "
        "Do not edit — re-run to update._\n\n"
        "Fixtures: normal mean-shift, lognormal tail-shift, 5% outlier injection, "
        "10% null injection. N=500 rows each.\n\n"
        f"{header}{sep}{body}"
    )
    Path("docs/benchmarks.md").write_text(out)
    print(f"docs/benchmarks.md updated — {len(rows)} detectors benchmarked")


if __name__ == "__main__":
    run_all()
```
